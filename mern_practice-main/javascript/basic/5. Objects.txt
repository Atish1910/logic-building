# Object creation using Functions:
    - Function creats an Object and Return 

    1. Factory Function
    2. Constructor Function

1. Factory Function:
    - Create an object and return as an output
    - Camel Case-notation is used - camelCaseHai

    function createRectangle (len, bre){
        // Return the Object 
        return rectangle = {
            length: len,
            breadth: bre,
            draw() {
                console.log("Drawing Rectangle");
            }
        };
    }

    // Create an object variable
    let rectangle_obj = createRectangle(5, 4);

2. Constructor Function:
    Pascal Notation is used - PascalNotationHai
    - Creates a new empty object -- New Keyword is used to create an Constructor

    function CreateRectangle (len, bre){
        // does not return anything
        this.length = len,
        this.breadth = bre,
        this.draw = function() {
            console.log("Drawing Rectangle");
        }
    }

    // Create an object variable - which will call Constructor Function using New Keyword
    let rectangle_obj = new CreateRectangle(5, 4);


    #---------------------------- return Non-Primitive
    # If we Try to return anything (Non-Primitive) then Output will be override with return value.
    function Person() {
        this.name = "Bob";
        return { age: 30 }; // Overrides the default return
    }

    const p = new Person();
    console.log(p); // { age: 30 }

    #----------------------------- return Primitive
    # If we return Primitive data then it will be ignored
    function Person() {
        this.name = "Bob";
        return "hello"; // Ignored
    }

    const p = new Person();
    console.log(p); // Person { name: 'Bob' }

# Deep Copy v/s Shallow Copy:
    1. Shallow Copy:
        - Shallow copy means copying only the first Object not the nestaed Object.
       
        let value_01 = {
            name:"Shubh",
            age:22,
            address:{
                city:"Pune",
                state:"Maharashtra"
            }
        };

        let value_02.{...value_01};
        // If we change nestaed Object in value_01 will change in Object of value_02 also.
    
    2. Deep Copy:
        - If changing nestaed Object in value_01 will not change in neasted Object of value_02.

# Iterating over the Object Properties:
    1. Using For-in Loop:

        const person = {name:"Shubham", age:22};
    
        for (let keys_01 in person){
            console.log(keys_01 + " : " + person[keys_01]);
        }

    2. using Object-keys and ForEach Loop:

        Object.keys(person).forEach((keys_01) =>{
            console.log(keys_01 + " : " + person[keys_01]);
        })
    
    3. using Object-values and forEach Loop.

        Object.values(person).forEach((values_01)=>{
            console.log(values_01);
        });

# Check Object Property is Present or not:

    var person = {name:"Shubham", age:22};

    1. Using "in" keyword/Operator:
        
        console.log("name" in person);    // true
    
    2. Using the hasOwnProperty() method:
        
        console.log(person.hasOwnProperty("name"));     // true
    
    3. Using comparing with undefined:

        console.log(person.name !== undefined);   // true

# Object.seal(object_name)
    - Allows to Read & Update Object Properties.
    - Can not add new Properties or delete existing properties.

    let my_obj = {"1":"first", "2":"second", "3":"third"};
    Object.seal(my_obj);

    // Now i can read & Update only the properties. Can't Add or Delete properties.

# Object.freeze(object_name)
    - Allows to Read Only the object properties.
    - Can not Add, Update or Delete the Property in Object.

    let my_obj = {"1":"first", "2":"second", "3":"third"};
    Object.freeze(my_obj);
